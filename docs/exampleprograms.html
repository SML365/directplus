4-function Calculator:
$ define variables
var({num1, num2, operator, result})
$ get user input
paste(input"enter first number: " => num1)
paste(input"enter second number: " => num2)
paste(input"enter operator {+|-|*|/}: " => operator)
$ perform calculation
if(@operator == "+") <<
  add(@num1, @num2 => result)
  >>
if(@operator == "-") <<
  sub(@num1, @num2 => result)
  >>
if(@operator == "*") <<
  mul(@num1, @num2 => result)
  >>
if(@operator == "/") <<
  div(@num1, @num2 => result)
  >>
paste(text"result: @result")
;
Fibonacci Sequence:
$ define variables
var({num1, num2, temp})
val(0 => num1)
val(1 => num2)
$ generate infinite fibonacci sequence
loop(0) <<
  paste(text"@num1")
  add(@num1, @num2 => temp)
  val(@num2 => num1)
  val(@temp => num2)
  >>
;
FizzBuzz:
$ define variables
var({temp1, temp2})
$ loop through numbers 1 to 100 (use built-in loop counter as number index)
loop(100, num) <<
  mod(@loop:num, 3 => temp1)
  mod(@loop:num, 5 => temp2)
  if(@temp1 == 0) <<
    if(@temp2 == 0) <<
      paste(text"fizzbuzz")
      >>
    else() <<
      paste(text"fizz")
      >>
    >>
    else() <<
      if(@temp2 == 0) <<
      paste(text"buzz")
      >>
    else() <<
      paste(text"@loop:num")
      >>
    >>
  >>
;
Reverse a String:
$ define variables
var({input, length, reversed, letter, index})
$ get user input
paste(input"enter a string: " => input)
length(@input => length)
var(index)
val(@length => index)
$ build the reversed string
loop(@length) <<
  letter(@index, @input => letter)
  concat(@reversed, @letter => reversed)
  sub(@index, 1 => index)
  >>
paste(text"reversed string: @reversed")
;
Linked List Example:
$ define variables
var({head, current_node, node_value, node_pointer, temp})
$ define linked list
var(head)
val("node1" => head)
var(node1)
val({"apple", "node2"} => node1)
var(node2)
val({"banana", "node3"} => node2)
var(node3)
val({"cherry", "null"} => node3)
$ traverse through list and print each item
var(current_node)
val(@head => current_node)
loop(0) <<
  run(/get_pointer[@@current_node] => node_pointer)
  item(@@current_node, 1 => node_value)
  paste(text"current item: @node_value")
  if(@node_pointer == "null") <<
    break()
    >>
  val(@node_pointer => current_node)
  >>
;
/get_pointer[node]
  item(@get_pointer:node, 2 => temp)
  /return[@temp]
;
