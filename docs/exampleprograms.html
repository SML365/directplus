4-function Calculator:
$ define variables
var({num1, num2, operator, result})
$ get user input
paste(input"enter first number: " => num1)
paste(input"enter second number: " => num2)
paste(input"enter operator {+|-|*|/}: " => operator)
$ perform calculation
if(@operator == "+") <<
  add(@num1, @num2 => result)
  >>
if(@operator == "-") <<
  sub(@num1, @num2 => result)
  >>
if(@operator == "*") <<
  mul(@num1, @num2 => result)
  >>
if(@operator == "/") <<
  div(@num1, @num2 => result)
  >>
paste(text"result: @result")
;
Fibonacci Sequence:
$ define variables
var({num1, num2, temp})
val(0 => num1)
val(1 => num2)
$ generate infinite fibonacci sequence
loop(0) <<
  paste(text"@num1")
  add(@num1, @num2 => temp)
  val(@num2 => num1)
  val(@temp => num2)
  >>
;
FizzBuzz:
$ define variables
var({temp1, temp2})
$ loop through numbers 1 to 100 (use built-in loop counter as number index)
loop(100, num) <<
  mod(@loop:num, 3 => temp1)
  mod(@loop:num, 5 => temp2)
  if(@temp1 == 0) <<
    if(@temp2 == 0) <<
      paste(text"fizzbuzz")
      >>
    else() <<
      paste(text"fizz")
      >>
    >>
    else() <<
      if(@temp2 == 0) <<
      paste(text"buzz")
      >>
    else() <<
      paste(text"@loop:num")
      >>
    >>
  >>
;
Reverse a String:
$ define variables
var({input, length, reversed, letter, index})
$ get user input
paste(input"enter a string: " => input)
length(@input => length)
var(index)
val(@length => index)
$ build the reversed string
loop(@length) <<
  letter(@index, @input => letter)
  concat(@reversed, @letter => reversed)
  sub(@index, 1 => index)
  >>
paste(text"reversed string: @reversed")
;
Linked List Example:
$ define variables
var(current_node)
$ define linked list
var({head, node1, node2, node3})
val("node1" => head)
val({"apple", "node2"} => node1)
val({"banana", "node3"} => node2)
val({"cherry", "null"} => node3)
$ traverse through the list and print each item
val(@head => current_node)
loop(0) <<
  paste(text"current value: @@current_node|:1")
  if("@@current_node|:2" == "null") <<
    break()
    >>
  val("@@current_node|:2" => current_node)
  >>
;
Program Structure:
It is recommended that you structure your programs accordingly. While Direct+ does not check for this structure, following this format will improve readability.
$ define metadata here (this is only required if you're writing an extension)
#name fruit_counter
#version 1.02
#author sml_365
#description an example program (fruit included)
#whatsnew fixed cherries not being counted properly
$ import any files here
use(__log)
$ define any variables here (group them by meaning and assign values if necessary)
var({temp1, temp2, temp3}) $ snake_case recommended with no underscores between numbers (ex: temp3 instead of temp_3)
var({apple_count, banana_count, cherry_count})
val(5 => temp1)
$ main code goes here
loop(10) <<
  paste(text"counting fruit...")
  run(/count_fruit) $ functions use the same casing as variables through functions from other files don't use casing
  >>
;
$ define any functions here
/count_fruit
  add(1, @apple_count => apple_count)
  add(2, @banana_count => banana_count)
  add(3, @cherry_count => cherry_count)
/return

Flagship Programs: Operating System
run(/boot)
run(/intro)
run(/repl)
;
/boot
  paste(text"directdos shell v1.0.0")
  use(__stringutils)
  use(__disk)
  use(__log)
  run(/__log:success["imported modules"])
  var({input, parsed_input})
  var({comm, arg1, arg2, arg3})
  var({tmp1, tmp2, elevated})
  val(false => elevated)
  run(/__log:success["defined variables"])
  run(/__log:warning["this next portion will take a while"])
  run(/__disk:init)
  run(/__disk:folder["bin"])
  run(/__disk:walk[{root, bin}])
  run(/__disk:file["cd", false])
  run(/__disk:file["read", false])
  run(/__disk:file["dplus", false])
  run(/__disk:file["file", false])
  run(/__disk:file["folder", false])
  run(/__disk:file["help", false])
  run(/__disk:file["write", false])
  run(/__disk:file["clear", false])
  run(/__disk:file["poweroff", false])
  run(/__disk:walk[{root}])
  run(/__log:success["mounted filesystem"])
/return
;
/intro
  run(/__log:info["run 'cd root\bin' and 'ls' to view commands"])
  run(/__log:info["run 'help' for additional help"])
/return
;
/repl
  loop(0) <<
    run(/get_input)
    turbo() <<
      run(/__stringutils:separate["@input", " "] => parsed_input)
      run(/map_parameters)
    >>
    numof(@__disk:root\bin, @comm => tmp1)
    item(@__disk:root\bin, @tmp1 => tmp2)
    if(@tmp1 == 0) <<
      run(/__log:error["can't execute @comm|: no such file or directory"])
    >>
    else() <<
      run(/@tmp2)
    >>
  >>
/return
;
/get_input
  paste(input"@__disk:full_path > " => input)
/return
;
/map_parameters
  turbo() <<
    if(@parsed_input:1 == "elevate") <<
      val(@parsed_input:2 => comm)
      val(@parsed_input:3 => arg1)
      val(@parsed_input:4 => arg2)
      val(@parsed_input:5 => arg3)
      val(true => elevated)
    >>
    else() <<
      val(@parsed_input:1 => comm)
      val(@parsed_input:2 => arg1)
      val(@parsed_input:3 => arg2)
      val(@parsed_input:4 => arg3)
      val(false => elevated)
    >>
  >>
/return
;
/cd
  turbo() <<
    run(/__stringutils:separate[@arg1, "\"] => tmp1)
    run(/__disk:walk[@tmp1])
  >>
/return
;
/read
  val("__disk:@__disk:full_path" => tmp1)
  run(/__log:success["@tmp1"])
/return
;
/file
  run(/__disk:file[@arg1, "."])
  run(/__log:success["file '@arg1|' is made in @__disk:full_path\@arg1"])
/return
;
/folder
  run(/__disk:folder[@arg1])
  run(/__log:success["folder '@arg1|' is made in @__disk:full_path\@arg1"])
/return
;
/write
  if(@elevated == true) <<
    paste(input">>> " => tmp1)
    run(/__disk:update["@arg1", @tmp1])
    run(/__log:success["updated @__disk:full_path\@arg1 to @tmp1"])
  >>
  else() <<
    run(/__log:warning["can't run without 'elevate' prefix"])
  >>
/return
;
/poweroff
  if(@elevated == true) <<
    clear()
    wait(0.5)
    run(/__log:info["running 'poweroff' participants"])
    wait(3)
    clear()
    exit()
  >>
  else() <<
    run(/__log:warning["can't run without 'elevate' prefix"])
  >>
/return
;
/dplus
  if(@elevated == true) <<
    exec(@arg1)
  >>
  else() <<
    run(/__log:warning["can't run without 'elevate' prefix"])
  >>
/return
;
/clear
  clear()
/return
;
/help
  color(148, 75, 0)
  paste(text"elevate - runs with privileges")
  paste(text"elevate poweroff - shuts down")
  paste(text"cd <absolute dir> - moves somewhere")
  paste(text"file <name> - makes a file")
  paste(text"folder <name> - makes a folder")
  paste(text"read - reads the contents of the current directory")
  paste(text"elevate write <name> - writes to a file %(a prompt will appear%)")
  paste(text"clear - clears the screen")
  paste(text"elevate dplus <code> - executes direct+ code")
  paste(text"help - shows this message")
  color(0, 100, 0)
/return
;
